import assert from 'node:assert/strict';
import {founderPlan} from '../src/lineage/body-inheritance.js';
import {makePlan} from '../src/lineage/body-plan.js';
import {tissueInventory} from '../src/organisms/tissue-structure.js';
import {createTissueLedger,tissueLedgerTotals,restoreTissueLedger} from '../src/organisms/tissue-ledger.js';
import {fundTissueConstruction,tissueComponentOwner,TISSUE_CONSTRUCTION_SCHEMA} from '../src/organisms/tissue-construction.js';
// All quantities/policy IDs below are synthetic. No tissue calibration or real
// embryo, anatomy route, body-gene scaling or gameplay material stock is implied.
const units={mass:{unit:'kg',numerator:'1',denominator:'1000000'},energy:{unit:'J',numerator:'1',denominator:'1000'}},body=founderPlan();
const base=()=>createTissueLedger({units,materials:['protein','water'],owners:{reserve:{materials:{protein:'10000',water:'10000'},energy:'10000'},heat:{materials:{},energy:'0'}}});
function request(plan=body){return{schema:TISSUE_CONSTRUCTION_SCHEMA,id:'construct:embryo:1',reason:'Synthetic explicit construction increment',organismId:'world-1:embryo-1',quantityPolicy:'synthetic-fixture/1',bodyPlan:structuredClone(plan),units:structuredClone(units),sources:{materialOwner:'reserve',energyOwner:'reserve',heatOwner:'heat'},modules:tissueInventory(plan).entries.map((e,i)=>({nodeId:e.nodeId,moduleId:'inherited-module-'+i,components:Object.fromEntries(e.components.map(c=>[c,{createOwner:true,materials:{protein:'2',water:'3'},energy:'5'}]))}))}}
const ledger=base(),opening=structuredClone(ledger),req=request(),original=structuredClone(req),funded=fundTissueConstruction(ledger,req);assert.ok(funded.ok);assert.deepEqual(ledger,opening);assert.deepEqual(req,original);assert.deepEqual(tissueLedgerTotals(funded.state),tissueLedgerTotals(ledger));assert.equal(funded.bindings.length,9,'Eight-node founder includes its one explicitly funded implicit head');const head=funded.bindings.find(b=>b.implicit);assert.equal(head.partId,'head-base');assert.equal(head.nodeId,'@implicit-head');const compartments=funded.bindings.reduce((n,b)=>n+Object.keys(b.owners).length,0);assert.equal(compartments,30);assert.equal(funded.state.owners.reserve.materials.protein,(10000n-2n*30n).toString());assert.equal(funded.state.owners.heat.energy,'150');
for(const binding of funded.bindings)for(const owner of Object.values(binding.owners)){assert.equal(funded.state.owners[owner].materials.protein,'2');assert.equal(funded.state.owners[owner].materials.water,'3')}
const retry=structuredClone(req);retry.modules.reverse();for(const m of retry.modules)m.components=Object.fromEntries(Object.entries(m.components).reverse());const replay=fundTissueConstruction(funded.state,retry);assert.ok(replay.ok&&replay.replayed);assert.equal(replay.state,funded.state);assert.deepEqual(replay.bindings,funded.bindings);const loaded=restoreTissueLedger(JSON.parse(JSON.stringify(funded.state)));assert.ok(fundTissueConstruction(loaded,req).replayed);
// Equal material payments do not license a different phenotype or policy under
// the same event identity. Padded and walking legs share component names.
const changed=request(makePlan(body.nodes.map(n=>({...n,partId:n.id==='root/limb.1'?'padded-leg':n.partId}))));assert.equal(fundTissueConstruction(funded.state,changed).reason,'event-conflict');assert.equal(fundTissueConstruction(funded.state,{...req,quantityPolicy:'different-fixture/1'}).reason,'event-conflict');
const poor=createTissueLedger({units,materials:['protein','water'],owners:{reserve:{materials:{protein:'1',water:'10000'},energy:'10000'},heat:{materials:{},energy:'0'}}}),poorBefore=structuredClone(poor),unfunded=fundTissueConstruction(poor,req);assert.equal(unfunded.reason,'insufficient-stock');assert.equal(unfunded.state,poor);assert.deepEqual(poor,poorBefore);assert.equal(Object.keys(poor.owners).length,2,'Failed construction cannot create component custody or a body');
for(const alter of [r=>r.modules.pop(),r=>delete r.modules[0].components.organ,r=>delete r.modules.find(m=>m.nodeId==='@implicit-head'),r=>r.modules.push({...r.modules[0],nodeId:'not-in-body'}),r=>r.modules[1].moduleId=r.modules[0].moduleId,r=>r.modules[0].components.unknown={createOwner:true,materials:{protein:'1'},energy:'1'},r=>r.modules[0].components.organ.energy=undefined,r=>r.modules[0].components.organ.materials.protein=1.5,r=>r.quantityPolicy=undefined]){const broken=request();alter(broken);const result=fundTissueConstruction(ledger,broken);assert.equal(result.ok,false);assert.equal(result.state,ledger);assert.deepEqual(ledger,opening)}
const noHead=request();noHead.modules=noHead.modules.filter(m=>m.nodeId!=='@implicit-head');assert.equal(fundTissueConstruction(ledger,noHead).reason,'missing-module-binding');
const badUnits=request();badUnits.units.mass.numerator=1n;assert.equal(fundTissueConstruction(ledger,badUnits).reason,'unit-contract-mismatch');const zero=request();for(const c of Object.values(zero.modules[0].components)){c.materials={};c.energy='0'}assert.equal(fundTissueConstruction(ledger,zero).reason,'empty-physical-module');
assert.equal(fundTissueConstruction(ledger,{...req,bodyPlan:{schema:body.schema}}).reason,'invalid-body');
// Tuples must not collide when IDs include separators; IDs are not socket paths.
assert.notEqual(tissueComponentOwner('a/b','c','muscle'),tissueComponentOwner('a','b/c','muscle'));
// A body with two optional sacs demands every additional declared compartment.
const withSacs=makePlan([...body.nodes.map(n=>({...n})),{partId:'storage-sac',parent:'root',socket:'flank.left'},{partId:'storage-sac',parent:'root',socket:'flank.right'}]),sacs=fundTissueConstruction(base(),request(withSacs));assert.ok(sacs.ok);assert.equal(sacs.bindings.length,11);assert.equal(sacs.state.owners.reserve.materials.protein,'9932');assert.equal(funded.state.owners.reserve.materials.protein,'9940');
console.log('Inactive anatomy construction: complete compartment/implicit-head ownership, explicit quantum increments, atomic funding, blueprint-bound retry, policy/units, synthetic organ differences and exact save replay PASS');
